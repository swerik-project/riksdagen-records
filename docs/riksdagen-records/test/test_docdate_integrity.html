riksdagen-records.test.test_docdate_integrity

Data integrity tests for protocol docDate guarantees.

These tests check release-blocking date guarantees for the records corpus: protocols must expose parseable TEI docDate values, protocol date spans should be short, same-chamber protocol date ranges must not move backwards in time, pre-1875 filename dates must be represented exactly in docDate, and docDate values should fall inside the expected Riksdag year for the folder.

The tests use the XML corpus in data/ as input and write one structured diagnostics table to test/results/docdate-integrity.csv when issues are found.

 58def parse_iso_date(value):
 59    """Return a datetime for YYYY-MM-DD strings, otherwise None."""
 60    try:
 61        return datetime.strptime(value, DATE_FORMAT).date()
 62    except:
 63        LOGGER.error(f"Could not parse date: {value}")
 64        return None

 79def expected_pre_1875_filename_date(path, year):
 80    """Infer the expected date from pre-1875 protocol filenames."""
 81    match = re.search(r"--(\d{4})$", Path(path).stem)
 82    if match is None:
 83        return None
 84    mmdd = match.group(1)
 85    return f"{year}-{mmdd[:2]}-{mmdd[2:]}"

 88def load_riksdag_year_ranges():
 89    """Load expected Riksdag-year ranges from the vendored test fixture."""
 90    global _RIKSDAG_YEAR_RANGES
 91    if _RIKSDAG_YEAR_RANGES is not None:
 92        return _RIKSDAG_YEAR_RANGES
 93
 94    if not RIKSDAG_YEAR_PATH.exists():
 95        raise FileNotFoundError(
 96            f"Missing {RIKSDAG_YEAR_PATH}; refresh it from "
 97            "../riksdagen-persons/data/riksdag-year.csv"
 98        )
 99
100    ranges = {}
101    LOGGER.info("Loading Riksdag year ranges from % s", RIKSDAG_YEAR_PATH)
102    rd_df = pl.read_csv(RIKSDAG_YEAR_PATH, try_parse_dates=True)
103    rd_df = rd_df.filter(pl.col("start").is_not_null() & pl.col("end").is_not_null())
104    for row in rd_df.to_dicts():
105        key = (str(row["parliament_year"]), row.get("chamber", None))
106        ranges.setdefault(key, []).append((row["start"], row["end"]))
107
108    _RIKSDAG_YEAR_RANGES = ranges
109    return _RIKSDAG_YEAR_RANGES

112def expected_riksdag_year_ranges(path, chamber):
113    """Return date ranges that are valid for a protocol's folder and chamber."""
114    folder = Path(path).parts[1]
115    ranges = load_riksdag_year_ranges()
116    chamber_code = CHAMBER_CODES.get(chamber)
117    expected = []
118    if chamber_code is not None:
119        expected.extend(ranges.get((folder, chamber_code), []))
120    expected.extend(ranges.get((folder, None), []))
121    return sorted(expected)

124def dates_outside_ranges(parsed_dates, expected_ranges):
125    return [
126        docdate
127        for docdate, parsed in parsed_dates
128        if not any(start <= parsed <= end for start, end in expected_ranges)
129    ]

132def collect_docdate_errors():
133    errors = []
134    previous_path = None
135    previous_chamber = None
136    previous_last_date = None
137
138    protocols = sorted(corpus_iterator("records", corpus_root="data"))
139    LOGGER.info("Checking docDate integrity for %s protocols", len(protocols))
140
141    for path in tqdm.tqdm(protocols):
142        metadata = infer_metadata(path)
143        chamber = metadata.get("chamber")
144        root, _ = parse_tei(path)
145        _, docdates = get_doc_dates(root)
146
147        if not docdates:
148            errors.append(
149                docdate_error(
150                    "missing_docdate",
151                    path,
152                    "missing docDate"
153                )
154            )
155        else:
156            parsed_dates = []
157            for docdate in docdates:
158                parsed = parse_iso_date(docdate)
159                if parsed is None:
160                    errors.append(
161                        docdate_error(
162                            "unparseable_docdate",
163                            path,
164                            "unparseable docDate",
165                            docdates=list(docdates),
166                        )
167                    )
168                else:
169                    parsed_dates.append(parsed)
170
171            if parsed_dates:
172                parsed_dates.sort()
173                first_date = parsed_dates[0]
174                last_date = parsed_dates[-1]
175
176                if len(parsed_dates) > 1:
177                    delta = last_date - first_date
178                    if delta.days > MAX_SPAN_DAYS:
179                        errors.append(
180                            docdate_error(
181                                "long_span",
182                                path,
183                                "protocol spans more than one week",
184                                first_docdate=first_date,
185                                last_docdate=last_date,
186                                span_days=delta.days,
187                            )
188                        )
189
190                if (
191                    previous_last_date is not None
192                    and previous_chamber == chamber
193                    and previous_last_date > first_date
194                ):
195                    errors.append(
196                        docdate_error(
197                            "range_overlap",
198                            path,
199                            "same-chamber date range overlap",
200                            previous_file=str(previous_path),
201                            previous_chamber=previous_chamber,
202                            chamber=chamber,
203                            previous_last_docdate=previous_last_date,
204                            first_docdate=first_date,
205                        )
206                    )
207
208                expected_ranges = expected_riksdag_year_ranges(path, chamber)
209                if not expected_ranges:
210                    errors.append(
211                        docdate_error(
212                            "missing_riksdag_year_range",
213                            path,
214                            "missing expected Riksdag-year range for folder/chamber",
215                            chamber=chamber,
216                        )
217                    )
218                else:
219                    outside_docdates = dates_outside_ranges(
220                        [(date.strftime(DATE_FORMAT), date) for date in parsed_dates],
221                        expected_ranges,
222                    )
223                    if outside_docdates:
224                        if len(outside_docdates) == len(parsed_dates):
225                            issue = "all docDates outside expected Riksdag year range"
226                        else:
227                            issue = "some docDates outside expected Riksdag year range"
228                        errors.append(
229                            docdate_error(
230                                "outside_riksdag_year_range",
231                                path,
232                                issue,
233                                docdates=list(docdates),
234                                expected_start=expected_ranges[0][0],
235                                expected_end=expected_ranges[-1][1],
236                                expected_ranges=join_ranges(expected_ranges),
237                                outside_docdates=list(outside_docdates),
238                            )
239                        )
240
241                year = metadata.get("year")
242                expected = None
243                if year and year < 1875:
244                    expected = expected_pre_1875_filename_date(path, year)
245                
246                if expected is not None:
247                    docdate_set = set(docdates)
248                    if expected not in docdate_set:
249                        errors.append(
250                            docdate_error(
251                                "filename_mismatch",
252                                path,
253                                "filename date not in docDate",
254                                expected_docdate=expected,
255                                docdates=list(docdates),
256                            )
257                        )
258                    elif len(docdate_set) > 1:
259                        errors.append(
260                            docdate_error(
261                                "filename_mismatch",
262                                path,
263                                "additional docDates beyond filename date",
264                                expected_docdate=expected,
265                                docdates=list(docdates),
266                            )
267                        )
268
269                previous_path = path
270                previous_chamber = chamber
271                previous_last_date = last_date
272
273    return errors

276def docdate_errors():
277    global _DOC_DATE_ERRORS
278
279    if _DOC_DATE_ERRORS is None:
280        errors = collect_docdate_errors()
281        _DOC_DATE_ERRORS = pl.DataFrame(errors, infer_schema_length=10000)
282        if len(_DOC_DATE_ERRORS) > 0:
283            RESULTS_PATH.parent.mkdir(parents=True, exist_ok=True)
284            _DOC_DATE_ERRORS = _DOC_DATE_ERRORS.sort(["file", "error_type"])
285            _DOC_DATE_ERRORS = _DOC_DATE_ERRORS.with_columns(pl.col("docdates").list.join("; "))
286            _DOC_DATE_ERRORS = _DOC_DATE_ERRORS.with_columns(pl.col("outside_docdates").list.join("; "))
287            _DOC_DATE_ERRORS.write_csv(RESULTS_PATH)
288
289    return _DOC_DATE_ERRORS

292class TestStringMethods(unittest.TestCase):
293    def test_protocols_have_parseable_docdates(self):
294        """Every protocol should have parseable docDate values."""
295        df = docdate_errors()
296        df_missing = df.filter(pl.col("error_type") == "missing_docdate")
297        df_unparseable = df.filter(pl.col("error_type") == "unparseable_docdate")
298        metadata_errors = len(df_missing) + len(df_unparseable)
299        self.assertEqual(metadata_errors, 0, 
300            f"{metadata_errors} protocol docDate metadata error(s); see {RESULTS_PATH}"
301        )
302
303
304    def test_protocol_date_spans_are_not_longer_than_one_week(self):
305        """A protocol should not span more than seven days."""
306        df_long = docdate_errors().filter(pl.col("error_type") == "long_span")
307        self.assertLessEqual(len(df_long), MAX_LONG_SPANS, 
308            f"{len(df_long)} protocol(s) span more than one week, exceeding baseline {MAX_LONG_SPANS}; see {RESULTS_PATH}"
309        )
310
311
312    def test_protocol_date_ranges_do_not_overlap_in_sequence(self):
313        """Adjacent same-chamber protocols should not move backwards in time."""
314        df_overlap =docdate_errors().filter(pl.col("error_type") == "range_overlap")
315        self.assertLessEqual(len(df_overlap), MAX_RANGE_OVERLAPS, 
316            f"{len(df_overlap)} protocol date range overlap(s), exceeding baseline {MAX_RANGE_OVERLAPS}; see {RESULTS_PATH}"
317        )
318
319
320    def test_pre_1875_filename_dates_match_docdates(self):
321        """Pre-1875 protocol filename dates should match the sole docDate."""
322        df_filename = docdate_errors().filter(pl.col("error_type") == "filename_mismatch")
323        self.assertLessEqual(len(df_filename), MAX_FILENAME_MISMATCHES, 
324            f"{len(df_filename)} pre-1875 filename/docDate mismatch(es), exceeding baseline {MAX_FILENAME_MISMATCHES}; see {RESULTS_PATH}"
325        )
326
327
328    def test_docdates_stay_inside_expected_riksdag_year_range(self):
329        """Protocol docDates should stay inside the Riksdag year implied by the folder."""
330        df_outside = docdate_errors().filter(pl.col("error_type") == "outside_riksdag_year_range")
331        self.assertLessEqual(len(df_outside), MAX_OUTSIDE_RIKSDAG_YEAR_RANGE, (
332            f"{len(df_outside)} protocol(s) have docDates outside the expected "
333            f"Riksdag year range, exceeding baseline "
334            f"{MAX_OUTSIDE_RIKSDAG_YEAR_RANGE}; see {RESULTS_PATH}"
335        ))
336
337
338    def test_protocols_have_expected_riksdag_year_ranges(self):
339        """Every protocol folder/chamber should have an expected Riksdag-year range."""
340        df_missing = docdate_errors().filter(pl.col("error_type") == "missing_riksdag_year_range")
341        self.assertLessEqual(len(df_missing), MAX_MISSING_RIKSDAG_YEAR_RANGE, (
342            f"{len(df_missing)} protocol(s) lack expected Riksdag-year ranges, "
343            f"exceeding baseline {MAX_MISSING_RIKSDAG_YEAR_RANGE}; see {RESULTS_PATH}"
344        ))
345
346
347    def test_known_199192_out_of_range_docdates_are_reported(self):
348        """Known issue-234 examples should be visible in range diagnostics until fixed."""
349        df_outside = docdate_errors().filter(pl.col("error_type") == "outside_riksdag_year_range")
350        rows = {
351            row["file"]: set(str(row["outside_docdates"]).split(";"))
352            for row in df_outside.select(["file", "outside_docdates"]).to_dicts()
353        }
354        missing = []
355        for path, known_docdates in KNOWN_199192_OUT_OF_RANGE_DOCDATES.items():
356            if Path(path).exists():
357                root, _ = parse_tei(path)
358                _, current_docdates = get_doc_dates(root)
359                if known_docdates.intersection(current_docdates) and path not in rows:
360                    missing.append(path)
361
362        self.assertFalse(missing, 
363            f"known issue-234 out-of-range docDate examples were not reported: {missing}; see {RESULTS_PATH}"
364        )
